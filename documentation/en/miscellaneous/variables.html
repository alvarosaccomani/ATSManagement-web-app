formatterDollar
Type : unknown
Default value : (value: number): string => value !== null && value !== undefined ? `$ ${value}`.replace(/\B(?=(\d{3})+(?!\d))/g, ',') : ''

Formatea un monto numérico en una cadena de moneda con prefijo de dólar y separadores de miles.
parserDollar
Type : unknown
Default value : (value: string): number => +value.replace(/\$\s?|(,*)/g, '')

src/app/core/interceptors/jwt.interceptor.ts

jwtInterceptor
Type : HttpInterceptorFn
Default value : (req, next) => { const sessionService = inject(SessionService); const session = sessionService.getCurrentSession(); let token = session?.token; if (!token) { try { const rawLocalStorage = typeof localStorage !== 'undefined' ? localStorage.getItem('session') : null; if (rawLocalStorage) { token = JSON.parse(rawLocalStorage)?.token; } } catch (e) { console.error('Error reading localStorage in jwtInterceptor', e); } } if (token) { req = req.clone({ setHeaders: { Authorization: `Bearer ${token}` } }); } return next(req); }

Interceptor HTTP que adjunta el token JWT Bearer a las solicitudes salientes.
