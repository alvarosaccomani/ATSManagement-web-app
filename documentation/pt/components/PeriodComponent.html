/**
 * Componente para crear y editar un período de facturación de tarjeta.
 * @description Gestiona el número de período de facturación (`YYYYMM`), fechas de inicio/fin/vencimiento, saldo anterior, intereses y estado.
 */
@Component({
  selector: 'app-period',
  standalone: true,
  imports: [
    CommonModule,
    FormsModule,
    ReactiveFormsModule,
    NzFormModule,
    NzInputModule,
    NzButtonModule,
    NzIconModule,
    NzSpinModule,
    NzDividerModule,
    NzSelectModule,
    NzDatePickerModule,
    NzInputNumberModule,
    NzTagModule
  ],
  templateUrl: './period.component.html',
  styleUrl: './period.component.scss'
})
export class PeriodComponent implements OnInit {
  /** Grupo de formulario reactivo para el período de facturación. */
  form: FormGroup;
  /** Indicador de si hay una solicitud en curso. */
  isLoading = false;
  /** Indicador de si el formulario se encuentra en modo de edición. */
  isEditing = false;
  /** UUID de la tarjeta padre obtenido de los parámetros de la ruta. */
  cardId: string | null = null;
  /** UUID del período que se está editando, o null al crear uno nuevo. */
  periodId: string | null = null;

  /** Lista de estados de período disponibles para el menú desplegable. */
  periodStatuses: StatusPeriodInterface[] = [];

  /** Lista de tarjetas del usuario para el menú desplegable. */
  cards: CardInterface[] = [];

  /**
   * Inicializa el componente PeriodComponent y construye el formulario reactivo.
   * @param fb Servicio FormBuilder de Angular.
   * @param cdr Servicio ChangeDetectorRef de Angular.
   * @param route ActivatedRoute para leer los parámetros de la ruta.
   * @param router Servicio Router de Angular.
   * @param _cardsService Servicio para obtener las tarjetas.
   * @param _periodsService Servicio para operaciones CRUD de períodos de facturación.
   * @param _statusPeriodsService Servicio para obtener los estados de períodos.
   * @param message Servicio de mensajes emergentes de Ng-Zorro.
   * @param _messageService Servicio personalizado de alertas modales.
   */
  constructor(
    private fb: FormBuilder,
    private cdr: ChangeDetectorRef,
    private route: ActivatedRoute,
    private router: Router,
    private _cardsService: CardsService,
    private _periodsService: PeriodsService,
    private _statusPeriodsService: StatusPeriodsService,
    private message: NzMessageService,
    private _messageService: MessageService
  ) {
    this.form = this.fb.group({
      usr_uuid: [{ value: '', disabled: true }],
      crd_uuid: [{ value: '', disabled: true }],
      per_uuid: [{ value: '', disabled: true }],
      per_periodnumber: ['', [Validators.required, Validators.pattern(/^\d{6}$/)]],
      per_startdate: [null, Validators.required],
      per_enddate: [null, Validators.required],
      per_duedate: [null, Validators.required],
      sper_uuid: ['1', Validators.required],
      per_previousbalance: [0, [Validators.min(0)]],
      per_interest: [0, [Validators.min(0)]]
    });
  }

  /**
   * Gancho del ciclo de vida ejecutado al inicializar el componente.
   * @returns void
   */
  ngOnInit(): void {
    this.cardId = this.route.snapshot.paramMap.get('crd_uuid');
    this.periodId = this.route.snapshot.paramMap.get('per_uuid');
    this.isEditing = !!this.periodId;
    this.loadCards();
    this.loadStatusPeriods();
    
    if (this.isEditing) {
      this.loadPeriod();
    } else {
      this.form.patchValue({
        usr_uuid: '7a067c90-bf5b-45d1-8f07-91b578e2d54d',
        crd_uuid: '1'
      })
      this.form.get('per_uuid')?.disable();
    }

    // Escuchar cambios en el número de período para formatear automáticamente
    this.form.get('per_periodnumber')?.valueChanges.subscribe(value => {
      if (value && typeof value === 'string') {
        // Limpiar y formatear (solo números, máximo 6 dígitos)
        const cleaned = value.replace(/\D/g, '').slice(0, 6);
        if (value !== cleaned) {
          this.form.get('per_periodnumber')?.setValue(cleaned, { emitEvent: false });
        }
      }
    });
  }

  /**
   * Carga un período de facturación existente por su ID y completa el formulario.
   * @returns void
   */
  loadPeriod(): void {
    this.isLoading = true;

    this._periodsService.getPeriodById('7a067c90-bf5b-45d1-8f07-91b578e2d54d', this.cardId!, this.periodId!).subscribe(
      (response: any) => {
        if(response.success) {
          console.info(response.data);
          const period = {
            ...response.data,
            per_startdate: new Date(response.data.per_startdate + 'T00:00:00'),
            per_enddate: new Date(response.data.per_enddate + 'T00:00:00'),
            per_duedate: new Date(response.data.per_duedate + 'T00:00:00')
          };

          this.form.get('per_uuid')?.enable();
          this.form.patchValue(period);
          this.form.get('per_uuid')?.disable();
          
          this.isLoading = false;
        } else {
          //this.status = 'error'
        }
      },
      (error: any) => {
        let errorMessage = <any>error;
        console.log(errorMessage);

        if(errorMessage != null) {
          //this.status = 'error'
        }
      }
    )
  }

  /**
   * Carga los estados de período disponibles desde el backend.
   * @returns void
   */
  loadStatusPeriods(): void {
    this.isLoading = true;

    this._statusPeriodsService.getStatusPeriods().subscribe(
      (response: any) => {
        if(response.success) {
          console.info(response.data);
          this.periodStatuses = response.data;
          
          this.isLoading = false;
        } else {
          //this.status = 'error'
        }
      },
      (error: any) => {
        let errorMessage = <any>error;
        console.log(errorMessage);

        if(errorMessage != null) {
          //this.status = 'error'
        }
      }
    )
  }

  /**
   * Carga las tarjetas del usuario disponibles desde el backend.
   * @returns void
   */
  loadCards(): void {
    this.isLoading = true;

    this._cardsService.getCards('7a067c90-bf5b-45d1-8f07-91b578e2d54d').subscribe(
      (response: any) => {
        if(response.success) {
          console.info(response.data);
          this.cards = response.data;
          
          this.isLoading = false;
        } else {
          //this.status = 'error'
        }
      },
      (error: any) => {
        let errorMessage = <any>error;
        console.log(errorMessage);

        if(errorMessage != null) {
          //this.status = 'error'
        }
      }
    )
  }

  /**
   * Valida el orden de las fechas y los controles del formulario, y luego crea o actualiza el período de facturación.
   * @returns void
   */
  onSubmit(): void {
    if (this.form.invalid) {
      Object.values(this.form.controls).forEach(control => {
        if (control.invalid) {
          control.markAsDirty();
          control.updateValueAndValidity({ onlySelf: true });
        }
      });
      this._messageService.error('Formulario Inválido', 'Por favor complete todos los campos obligatorios');
      return;
    }

    // Validar que la fecha de inicio sea anterior a la fecha de fin
    const startDate = this.form.get('per_startdate')?.value;
    const endDate = this.form.get('per_enddate')?.value;
    const dueDate = this.form.get('per_duedate')?.value;

    if (startDate && endDate && startDate >= endDate) {
      this._messageService.error('Error de Fechas', 'La fecha de inicio debe ser anterior a la fecha de fin');
      return;
    }

    if (endDate && dueDate && endDate >= dueDate) {
      this._messageService.error('Error de Fechas', 'La fecha de fin debe ser anterior a la fecha de vencimiento');
      return;
    }

    this.isLoading = true;

    if(!this.isEditing) {
      this._periodsService.savePeriod(this.form.getRawValue()).subscribe({
        next: response => {
          this.isLoading = false;
          if(response.success) {
            this._messageService.success(
              'Período Creado', 
              'El período se agregó correctamente.',
              () => {
                this.router.navigate(['/admin/user/periods']);
              }
            );
          } else {
            this._messageService.error('Error', response.message || 'No se pudo guardar el período');
          }
        },
        error: error => {
          this.isLoading = false;
          let errorMessage = <any>error;
          console.error(errorMessage);
          this._messageService.error('Error', errorMessage.error?.error || 'Hubo un problema al guardar el período');
        }
      });
    } else {
      this._periodsService.updatePeriod(this.form.getRawValue()).subscribe({
        next: response => {
          this.isLoading = false;
          if(response.success) {
            this._messageService.success(
              'Período Actualizado', 
              'El período se actualizó correctamente.',
              () => {
                this.router.navigate(['/admin/user/periods']);
              }
            );
          } else {
            this._messageService.error('Error', response.message || 'No se pudo actualizar el período');
          }
        },
        error: error => {
          this.isLoading = false;
          let errorMessage = <any>error;
          console.error(errorMessage);
          this._messageService.error('Error', errorMessage.error?.error || 'Hubo un problema al actualizar el período');
        }
      });
    }
  }

  /**
   * Cancela la operación y navega de regreso a la lista de períodos.
   * @returns void
   */
  onCancel(): void {
    this.router.navigate(['/admin/user/periods']);
  }

  /**
   * Obtiene el mensaje de error de validación para un control del formulario.
   * @param controlName Nombre del control del formulario.
   * @returns Mensaje de error de validación o null.
   */
  getErrorMessage(controlName: string): string | null {
    const control = this.form.get(controlName);
    if (control?.hasError('required')) {
      return 'Este campo es obligatorio';
    }
    if (control?.hasError('pattern')) {
      if (controlName === 'per_periodnumber') {
        return 'Ingrese exactamente 6 dígitos (YYYYMM)';
      }
      return 'Formato inválido';
    }
    if (control?.hasError('min')) {
      return 'El valor mínimo es ' + control.errors?.['min'].min;
    }
    return null;
  }

  /** Función formateadora de visualización de moneda. */
  formatterDollar = formatterDollar;
  /** Función analizadora de entrada de moneda. */
  parserDollar = parserDollar;
  /** Alias para la función formateadora de visualización de moneda. */
  currencyFormatter = formatterDollar;
  /** Alias para la función analizadora de entrada de moneda. */
  currencyParser = parserDollar;

  /**
   * Devuelve el color de estado de insignia de Ng-Zorro para un ID de estado de período.
   * @param statusId Identificador del estado.
   * @returns Cadena de estado de la insignia.
   */
  getStatusColor(statusId: string): string {
    switch (statusId) {
      case '1': return 'success';    // Abierto
      case '2': return 'default';    // Cerrado
      case '3': return 'processing'; // Pagado
      case '4': return 'error';      // Vencido
      default: return 'default';
    }
  }

  /**
   * Devuelve el nombre del ícono de Ng-Zorro para un ID de estado de período.
   * @param statusId Identificador del estado.
   * @returns Cadena con el nombre del ícono.
   */
  getStatusIcon(statusId: string): string {
    switch (statusId) {
      case '1': return 'unlock';            // Abierto
      case '2': return 'lock';              // Cerrado
      case '3': return 'check-circle';      // Pagado
      case '4': return 'exclamation-circle'; // Vencido
      default: return 'info-circle';
    }
  }

  /**
   * Resuelve la etiqueta de visualización para un ID de estado de período.
   * @param statusId Identificador del estado.
   * @returns Cadena con la etiqueta del estado.
   */
  getStatusLabel(statusId: string): string {
    const status = this.periodStatuses.find(s => s.sper_uuid === statusId);
    return status ? status.sper_name : 'Desconocido';
  }

  /**
   * Formatea una cadena de período de 6 dígitos (`YYYYMM`) como `MM/YYYY`.
   * @param value Cadena de período de 6 dígitos sin procesar.
   * @returns Cadena de período formateada.
   */
  formatPeriodNumber(value: string): string {
    if (!value || value.length !== 6) return value;
    const year = value.substring(0, 4);
    const month = value.substring(4, 6);
    return `${month}/${year}`;
  }
}
